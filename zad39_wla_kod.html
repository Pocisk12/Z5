    <script>
        function fun_wla() {
            var x_wla = prompt("Podaj liczbę (1 lub 2) lub nazwę koloru (czerwony lub niebieski):", "");

            if (x_wla != null) {
                
                if (x_wla == "1" || x_wla.toLowerCase() == "czerwony") {
                    document.write('<font size="7" color="red">wczytałeś kolor czerwony</font><br />');
                } 
                
                else if (x_wla == "2" || x_wla.toLowerCase() == "niebieski") {
                    document.write('<font size="1" color="blue">wczytałeś kolor niebieski</font><br />');
                } 
                
                else {
                    document.write("Nie rozpoznano wyboru.");
                }
            }
        }
    </script>
